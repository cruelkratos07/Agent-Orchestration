# Phase 9: model providers. Install ONE of these - whichever key you have.
#
# The project runs with none of them (deterministic keyword mode), so this file
# is entirely optional. Pick a provider with LLM_PROVIDER, or just set a key and
# it is auto-detected. LLM_MODEL overrides the model name.
#
#   Gemini   - free tier, least friction        https://aistudio.google.com/apikey
#     pip install langchain-google-genai
#     $env:GEMINI_API_KEY = "..."
#
#   Groq     - free tier, very fast             https://console.groq.com/keys
#     pip install langchain-groq
#     $env:GROQ_API_KEY = "..."
#
#   Anthropic - paid                            https://console.anthropic.com
#     pip install langchain-anthropic
#     $env:ANTHROPIC_API_KEY = "..."
#
#   Anything OpenAI-compatible (OpenRouter, Cerebras, Together, local Ollama):
#     pip install langchain-openai
#     $env:OPENAI_API_KEY = "..."; $env:OPENAI_BASE_URL = "https://..."
#
# Uncomment the one you want, or install it directly with pip.
